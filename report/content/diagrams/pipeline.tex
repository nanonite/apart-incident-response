% From API log-probabilities to per-agent and system entropy.
\begin{tikzpicture}[x=1cm, y=1cm]
  \tikzset{pbox/.style={box, text width=3.4cm, minimum height=1.35cm}}
  \node[pbox, line width=0.8pt] (lp) at (0,0) {\textbf{Top-20 log-}\\\textbf{probabilities} $\ell_1,\dots,\ell_{20}$\\ for every generated token};
  % token path
  \node[pbox] (tok) at (4.25,1.05) {\textbf{Token entropy}\\ $H_t$ over the renormalised\\ top 20, bound $H^{\mathrm{low}}_t$,\\ coverage};

  \draw[flow] (mix) -- (sys);
  \node[font=\scriptsize\bfseries, anchor=south west, text=logink] at ($(tok.north west)+(0,0.05)$) {per agent};
  \node[font=\scriptsize\bfseries, anchor=north west, text=logink] at ($(q.south west)+(0,-0.05)$) {two-agent system};
